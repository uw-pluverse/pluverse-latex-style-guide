% ---------------------------------------------------------------------------
% URLs and hyperlinks.
% IEEEtran loads neither hyperref nor T1 font encoding. Without hyperref,
% \url{...} is plain text and PDF viewers guess the link, truncating it at a
% line break; with the default OT1 encoding, `_' is not a real glyph, so the
% guessed link (and copy-paste) also breaks at underscores.
% acmart already loads both, so we only add them when missing
% (T1 is skipped for acmart; see the note in pluverse-preamble.tex).
% hyperref must be loaded before cleveref.
% ---------------------------------------------------------------------------
\makeatletter
\@ifclassloaded{acmart}{}{\usepackage[T1]{fontenc}}
\@ifpackageloaded{hyperref}{}{\usepackage[hidelinks]{hyperref}}
\makeatother

\usepackage{cleveref} % This package must be loaded in the end.


\Crefname{algocf}{Algorithm}{Algorithms}
\crefname{algocf}{Algorithm}{Algorithms}

\Crefname{algorithm}{Algorithm}{Algorithms}
\crefname{algorithm}{Algorithm}{Algorithms}

\crefname{appendix}{Appendix}{Appendices}
\Crefname{appendix}{Appendix}{Appendices}

\Crefname{figure}{Figure}{Figures}
\crefname{figure}{Figure}{Figures}

\crefname{listing}{Listing}{Listings}
\Crefname{listing}{Listing}{Listings}

\Crefname{table}{Table}{Tables}
\crefname{table}{Table}{Tables}

\crefname{thm}{Theorem}{Theorems}
\Crefname{thm}{Theorem}{Theorems}

\crefname{definition}{Definition}{Definitions}
\Crefname{definition}{Definition}{Definitions}

\crefname{lemma}{Lemma}{Lemmas}
\Crefname{lemma}{Lemma}{Lemmas}

\crefname{corollary}{Corollary}{Corollaries}
\Crefname{corollary}{Corollary}{Corollaries}

\crefname{proof}{Proof}{Proofs}
\Crefname{proof}{Proof}{Proofs}

\crefname{equation}{Equation}{Equations}
\Crefname{equation}{Equation}{Equations}

% https://tex.stackexchange.com/questions/81634/cleveref-configure-symbol-for-all-sectioning-types-once-time
\crefformat{chapter}{\S~#2#1#3}
\crefmultiformat{chapter}{\S\S~#2#1#3}{ and~#2#1#3}{, #2#1#3}{, and~#2#1#3}

\crefformat{section}{\S~#2#1#3}
\crefmultiformat{section}{\S\S~#2#1#3}{ and~#2#1#3}{, #2#1#3}{, and~#2#1#3}

% ---------------------------------------------------------------------------
% algorithm2e + hyperref: algorithm2e steps line numbers with
% \hyper@refstepcounter, which bypasses cleveref, so \cref{<line label>}
% prints "Algorithm k" instead of "Line k". Re-step the counter with
% \refstepcounter so cleveref sees the `line' type.
% ---------------------------------------------------------------------------
\makeatletter
\@ifpackageloaded{algorithm2e}{%
  \renewcommand{\algocf@nl@sethref}[1]{%
    \renewcommand{\theHAlgoLine}{\thealgocfproc.#1}%
    \addtocounter{AlgoLine}{-1}\refstepcounter{AlgoLine}\gdef\@currentlabel{#1}%
  }%
}{}
\makeatother
% \cref (within a sentence) -> "line 10"; \Cref (start of a sentence) -> "Line 10".
\crefname{line}{line}{lines}
\Crefname{line}{Line}{Lines}
\crefrangeformat{line}{line~#3#1#4--#5#2#6}
\Crefrangeformat{line}{Line~#3#1#4--#5#2#6}
